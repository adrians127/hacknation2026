cmake_minimum_required(VERSION 3.28)
project(httpx-worker VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
add_executable(httpx-worker src/address.cpp src/job.cpp src/main.cpp src/process.cpp src/result.cpp)
target_include_directories(httpx-worker PRIVATE ${NLOHMANN_JSON_INCLUDE_DIR})
target_link_libraries(httpx-worker PRIVATE OpenSSL::Crypto Threads::Threads)
target_compile_definitions(httpx-worker PRIVATE HTTPX_WORKER_VERSION="${PROJECT_VERSION}")
target_compile_options(httpx-worker PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion -Werror)
install(TARGETS httpx-worker RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME httpx-worker-integration COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_worker.py $<TARGET_FILE:httpx-worker>)
  endif()
endif()
